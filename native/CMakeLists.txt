cmake_minimum_required(VERSION 3.25)
project(CodexLanguageFixNative VERSION 2.1.5 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC v143 are required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/json.hpp" JSON_HEADER_SHA256)
if(NOT JSON_HEADER_SHA256 STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "nlohmann/json 3.12.0 header does not match the pinned release SHA-256")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(clf_core STATIC src/core_unicode.cpp src/core_text.cpp src/geometry.cpp src/localization.cpp)
target_include_directories(clf_core PUBLIC include PRIVATE third_party)
target_compile_definitions(clf_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(clf_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(clf_core PUBLIC bcrypt icu)
add_library(clf_platform STATIC src/composer.cpp src/backend.cpp src/settings.cpp)
target_include_directories(clf_platform PRIVATE third_party)
target_link_libraries(clf_platform PUBLIC clf_core user32 shell32 ole32 oleaut32 uiautomationcore normaliz advapi32 shlwapi xmllite)
add_library(clf_controller STATIC src/controller.cpp)
target_link_libraries(clf_controller PUBLIC clf_core)
add_executable(CodexLanguageFix WIN32 src/main.cpp src/desktop.cpp resources/application.rc)
target_link_libraries(CodexLanguageFix PRIVATE clf_platform clf_controller comctl32 gdi32)
target_link_options(CodexLanguageFix PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/application.manifest)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE clf_core)
  add_test(NAME native_core COMMAND core_tests)
  add_executable(native_differential tests/differential.cpp)
  target_include_directories(native_differential PRIVATE third_party)
  target_link_libraries(native_differential PRIVATE clf_platform)
  add_executable(controller_tests tests/controller_tests.cpp)
  target_link_libraries(controller_tests PRIVATE clf_controller)
  add_test(NAME native_controller COMMAND controller_tests)
  add_executable(fake_codex_server tests/fake_server.cpp)
  target_include_directories(fake_codex_server PRIVATE third_party)
  target_link_libraries(fake_codex_server PRIVATE clf_core)
  add_executable(backend_tests tests/backend_tests.cpp src/backend.cpp src/settings.cpp)
  target_include_directories(backend_tests PRIVATE third_party)
  target_compile_definitions(backend_tests PRIVATE CLF_TESTING)
  target_link_libraries(backend_tests PRIVATE clf_core user32 shell32 ole32 oleaut32 advapi32 shlwapi xmllite)
  add_test(NAME native_backend COMMAND backend_tests $<TARGET_FILE:fake_codex_server>)
  set_tests_properties(native_backend PROPERTIES TIMEOUT 45)
  add_executable(live_backend_probe tests/live_backend_probe.cpp)
  target_include_directories(live_backend_probe PRIVATE third_party)
  target_link_libraries(live_backend_probe PRIVATE clf_platform)
  add_test(NAME native_backend_live COMMAND live_backend_probe)
  set_tests_properties(native_backend_live PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180 LABELS "live")
  add_executable(uia_probe tests/uia_probe.cpp src/composer.cpp)
  target_include_directories(uia_probe PRIVATE third_party)
  target_link_libraries(uia_probe PRIVATE clf_core user32 shell32 ole32 oleaut32 uiautomationcore normaliz)
  set_tests_properties(native_core native_controller PROPERTIES TIMEOUT 30)
endif()
install(TARGETS CodexLanguageFix RUNTIME DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.LICENSE.MIT)
install(FILES third_party/dotnet/LICENSE.TXT DESTINATION licenses RENAME dotnet-runtime.LICENSE.TXT)
install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
